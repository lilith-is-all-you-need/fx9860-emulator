cmake_minimum_required(VERSION 3.15)
project(FxStudioCppRuntime C CXX)
include(GenerateG1A)
find_package(Gint 2.9 REQUIRED)
add_executable(cpp_runtime main.cpp)
target_compile_features(cpp_runtime PRIVATE cxx_std_17)
target_compile_options(cpp_runtime PRIVATE -Wall -Wextra)
target_link_libraries(cpp_runtime stdc++ Gint::Gint)
target_link_options(cpp_runtime PRIVATE "-Wl,-Map,cpp-runtime.map")
generate_g1a(TARGET cpp_runtime OUTPUT "CppRuntime.g1a" NAME "C++ RT" INTERNAL "@FXCPPRT")
